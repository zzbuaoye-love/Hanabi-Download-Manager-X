    CACHE STRING "" FORCE)

      STRING "Flutter build mode" FORCE)

# Use Unicode for all projects.

# Compilation settings that should be applied to most targets.
#
# Be cautious about adding new options here, as plugins use this function by
# default. In most cases, you should add new options to specific targets instead
# of modifying this function.
function(APPLY_STANDARD_SETTINGS TARGET)
  target_compile_features(${TARGET} PUBLIC cxx_std_17)
  target_compile_options(${TARGET} PRIVATE /W4 /WX /wd"4100")
  # 显式声明源码与执行字符集都是 UTF-8。
  #
  # 缺了这个，MSVC 会按系统 ANSI 代码页解释源文件；在中文（CP936）机器上，
  # 任何一个非 ASCII 字符——哪怕只是注释里的中文——都会触发 C4819，
  # 而上面的 /WX 又把警告升级成错误，于是构建必然失败。
  # 这一类问题只在中文环境暴露，英文环境下完全看不出来。
  target_compile_options(${TARGET} PRIVATE /utf-8)
  target_compile_options(${TARGET} PRIVATE /EHsc)
  target_compile_definitions(${TARGET} PRIVATE "_HAS_EXCEPTIONS=0")
  target_compile_definitions(${TARGET} PRIVATE "$<$<CONFIG:Debug>:_DEBUG>")
endfunction()

# === Installation ===
# Support files are copied into place next to the executable, so that it can
# run in place. This is done instead of making a separate bundle (as on Linux)
# so that building and running from within Visual Studio will work.
set(BUILD_BUNDLE_DIR "$<TARGET_FILE_DIR:${BINARY_NAME}>")
# Make the "install" step default, as it's required to run.
set(CMAKE_VS_INCLUDE_INSTALL_TO_DEFAULT_BUILD 1)
if(CMAKE_INSTALL_PREFIX_INITIALIZED_TO_DEFAULT)
  set(CMAKE_INSTALL_PREFIX "${BUILD_BUNDLE_DIR}" CACHE PATH "..." FORCE)
endif()

set(INSTALL_BUNDLE_DATA_DIR "${CMAKE_INSTALL_PREFIX}/data")
set(INSTALL_BUNDLE_LIB_DIR "${CMAKE_INSTALL_PREFIX}")
set(CLOSE_STALE_BUNDLE_SCRIPT "${CMAKE_CURRENT_SOURCE_DIR}/runner/close_stale_bundle.ps1")

install(CODE "
  if(CMAKE_INSTALL_CONFIG_NAME MATCHES \"^([Dd][Ee][Bb][Uu][Gg])$\")
    set(bundle_executable \"${PROJECT_BINARY_DIR}/runner/Debug/${BINARY_NAME}.exe\")
  elseif(CMAKE_INSTALL_CONFIG_NAME MATCHES \"^([Pp][Rr][Oo][Ff][Ii][Ll][Ee])$\")
    set(bundle_executable \"${PROJECT_BINARY_DIR}/runner/Profile/${BINARY_NAME}.exe\")
  elseif(CMAKE_INSTALL_CONFIG_NAME MATCHES \"^([Rr][Ee][Ll][Ee][Aa][Ss][Ee])$\")
    set(bundle_executable \"${PROJECT_BINARY_DIR}/runner/Release/${BINARY_NAME}.exe\")
  endif()

  if(DEFINED bundle_executable AND EXISTS \"\${bundle_executable}\")
    execute_process(
      COMMAND powershell -NoProfile -NonInteractive -ExecutionPolicy Bypass
        -File \"${CLOSE_STALE_BUNDLE_SCRIPT}\"
        -ExecutablePath \"\${bundle_executable}\"
      RESULT_VARIABLE close_result
      OUTPUT_VARIABLE close_output
      ERROR_VARIABLE close_error
    )
    if(NOT close_result EQUAL 0)
      message(WARNING \"Failed to close a stale build output instance for \${bundle_executable}: \${close_output}\${close_error}\")
    endif()
  endif()
" COMPONENT Runtime)

install(TARGETS ${BINARY_NAME} RUNTIME DESTINATION "${CMAKE_INSTALL_PREFIX}"
  COMPONENT Runtime)

install(FILES "${FLUTTER_ICU_DATA_FILE}" DESTINATION "${INSTALL_BUNDLE_DATA_DIR}"
  COMPONENT Runtime)

install(FILES "${FLUTTER_LIBRARY}" DESTINATION "${INSTALL_BUNDLE_LIB_DIR}"
  COMPONENT Runtime)

if(PLUGIN_BUNDLED_LIBRARIES)
  install(FILES "${PLUGIN_BUNDLED_LIBRARIES}"
    DESTINATION "${INSTALL_BUNDLE_LIB_DIR}"
    COMPONENT Runtime)
endif()

     DESTINATION "${INSTALL_BUNDLE_LIB_DIR}"

# Fully re-copy the assets directory on each build to avoid having stale files
# from a previous install.
set(FLUTTER_ASSET_DIR_NAME "flutter_assets")
install(CODE "
  file(REMOVE_RECURSE \"${INSTALL_BUNDLE_DATA_DIR}/${FLUTTER_ASSET_DIR_NAME}\")
  " COMPONENT Runtime)
install(DIRECTORY "${PROJECT_BUILD_DIR}/${FLUTTER_ASSET_DIR_NAME}"
  DESTINATION "${INSTALL_BUNDLE_DATA_DIR}" COMPONENT Runtime)

# NeoNSF is built outside Flutter, but a plain `flutter build windows` must not
# retain a stale sidecar from an older install. Always clear the bundle slot,
# then copy the current build when it exists.
set(NEONSF_BUILD_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../build/neonsf/win-x64")
set(NEONSF_BUNDLE_DIR
  "${INSTALL_BUNDLE_DATA_DIR}/zzbuaoye_assets/neonsf")
install(CODE "
  file(REMOVE_RECURSE \"${NEONSF_BUNDLE_DIR}\")
  " COMPONENT Runtime)
if(EXISTS "${NEONSF_BUILD_DIR}/HanabiNeoNSF.exe")
  install(DIRECTORY "${NEONSF_BUILD_DIR}/"
    DESTINATION "${NEONSF_BUNDLE_DIR}"
    COMPONENT Runtime)
endif()
